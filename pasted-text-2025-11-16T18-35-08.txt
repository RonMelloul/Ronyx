const getCalculatedSignatureStatus = (owner) => {
    if (!owner) return 'missing';
    const signatureDocs = [
        ...Object.values(owner.legalSignatures || {}),
        ...Object.values(owner.requiredDocuments || {}).map(doc => doc.status)
    ];

    const positive = ['V', 'signed', 'ok'];
    const negative = ['X', 'missing', 'rejected'];
    const pending = ['pending'];

    if (signatureDocs.length === 0) return 'pending';
    if (signatureDocs.every(s => positive.includes(s))) return 'signed';
    if (signatureDocs.some(s => negative.includes(s))) return 'rejected';
    if (signatureDocs.some(s => pending.includes(s))) return 'pending';
    return 'pending';
};

// --- SPECIAL MAJORITY CALCULATOR (Based on Israeli Law) ---
const SpecialMajorityCalculator = {
    // Constants for Agreement Majority
    AGREEMENT_CLUSTER_APARTMENT_THRESHOLD: 2 / 3,
    AGREEMENT_BUILDING_APARTMENT_THRESHOLD_GENERAL: 3 / 5,
    AGREEMENT_BUILDING_COMMON_PROPERTY_THRESHOLD: 1 / 2,

    // Constants for Economic Feasibility Majority
    FEASIBILITY_BUILDING_APARTMENT_THRESHOLD: 2 / 5,
    FEASIBILITY_BUILDING_COMMON_PROPERTY_THRESHOLD: 2 / 5,

    // Section 4 constants (for owners with large holdings)
    SECTION_4_OWNERSHIP_LIMIT: 0.30,
    SECTION_4_EXCESS_FACTOR: 1 / 3,

    EPSILON: 1e-9,

    analyze(buildingsData, scenarioSignedOwners = new Set()) {
        const allClusterUnits = buildingsData.flatMap(b => b.units);
        if (allClusterUnits.length === 0) return this.getEmptyResults();

        let totalClusterSignedApartments = 0;
        const buildingDetails = {};

        buildingsData.forEach(building => {
            const totalApartments = building.units.length;
            const uniqueOwners = new Set(building.units.flatMap(u => u.owners.map(o => o.idNumber))).size;

            const signedUnitsInBuilding = new Set();
            let signedCommonProperty = 0;

            building.units.forEach(unit => {
                const isSigned = unit.owners.some(owner => scenarioSignedOwners.has(owner.idNumber));
                if (isSigned) {
                    signedUnitsInBuilding.add(unit.id);
                    signedCommonProperty += (unit.legal.commonPropertyPercentage || 0);
                }
            });

            totalClusterSignedApartments += signedUnitsInBuilding.size;

            const totalCommonProperty = building.units.reduce((sum, unit) => sum + (unit.legal.commonPropertyPercentage || 0), 0);

            // --- Agreement Majority Calculation ---
            let agreementApartmentCheckMet = false;
            if (totalApartments >= 6) {
                agreementApartmentCheckMet = (signedUnitsInBuilding.size / totalApartments) >= (this.AGREEMENT_BUILDING_APARTMENT_THRESHOLD_GENERAL - this.EPSILON);
            } else if ((totalApartments === 4 || totalApartments === 5) && uniqueOwners > 2) {
                agreementApartmentCheckMet = signedUnitsInBuilding.size >= 3;
            }
            const agreementCommonPropertyCheckMet = totalCommonProperty > 0 && (signedCommonProperty / totalCommonProperty) > (this.AGREEMENT_BUILDING_COMMON_PROPERTY_THRESHOLD - this.EPSILON);

            // --- Feasibility Majority Calculation ---
            const feasibilityApartmentCheckMet = (signedUnitsInBuilding.size / totalApartments) >= (this.FEASIBILITY_BUILDING_APARTMENT_THRESHOLD - this.EPSILON);
            const feasibilityCommonPropertyCheckMet = totalCommonProperty > 0 && (signedCommonProperty / totalCommonProperty) >= (this.FEASIBILITY_BUILDING_COMMON_PROPERTY_THRESHOLD - this.EPSILON);

            buildingDetails[building.id] = {
                isValidForCalc: totalApartments >= 4,
                signedApartments: signedUnitsInBuilding.size,
                totalApartments,
                signedCommonProperty,
                totalCommonProperty,
                agreement: {
                    apartment: { met: agreementApartmentCheckMet, actual: signedUnitsInBuilding.size, required: totalApartments < 6 ? 3 : Math.ceil(totalApartments * this.AGREEMENT_BUILDING_APARTMENT_THRESHOLD_GENERAL) },
                    commonProperty: { met: agreementCommonPropertyCheckMet, actual: signedCommonProperty, required: totalCommonProperty * this.AGREEMENT_BUILDING_COMMON_PROPERTY_THRESHOLD },
                    overallMet: totalApartments >= 4 && agreementApartmentCheckMet && agreementCommonPropertyCheckMet
                },
                feasibility: {
                    apartment: { met: feasibilityApartmentCheckMet, actual: signedUnitsInBuilding.size, required: Math.ceil(totalApartments * this.FEASIBILITY_BUILDING_APARTMENT_THRESHOLD) },
                    commonProperty: { met: feasibilityCommonPropertyCheckMet, actual: signedCommonProperty, required: totalCommonProperty * this.FEASIBILITY_BUILDING_COMMON_PROPERTY_THRESHOLD },
                    overallMet: totalApartments >= 4 && feasibilityApartmentCheckMet && feasibilityCommonPropertyCheckMet
                }
            };
        });

        const validBuildingsForCalc = buildingsData.filter(b => (buildingDetails[b.id]?.isValidForCalc));
        const agreementClusterApartmentMet = allClusterUnits.length > 0 ? (totalClusterSignedApartments / allClusterUnits.length) >= (this.AGREEMENT_CLUSTER_APARTMENT_THRESHOLD - this.EPSILON) : false;
        const allValidBuildingsAgreementMet = validBuildingsForCalc.length > 0 && validBuildingsForCalc.every(b => buildingDetails[b.id].agreement.overallMet);

        return {
            cluster: {
                agreement: {
                    apartment: { met: agreementClusterApartmentMet, actual: totalClusterSignedApartments, required: Math.ceil(allClusterUnits.length * this.AGREEMENT_CLUSTER_APARTMENT_THRESHOLD) },
                    overallMet: agreementClusterApartmentMet && allValidBuildingsAgreementMet
                },
                totalApartments: allClusterUnits.length,
            },
            buildings: buildingDetails
        };
    },
    getEmptyResults() {
        return { cluster: { agreement: { apartment: {}, overallMet: false }, totalApartments: 0 }, buildings: {} };
    }
};


// --- UTILITY FUNCTIONS ---
const getNestedValue = (obj, path) => {
    if (!path || !obj) return undefined;
    return path.split('.').reduce((o, key) => (o && o[key] !== undefined) ? o[key] : undefined, obj);
}

const setNestedValue = (obj, path, value) => {
    const keys = path.split('.');
    const lastKey = keys.pop();
    const target = keys.reduce((o, key) => {
        if (o[key] === undefined || o[key] === null || typeof o[key] !== 'object') {
            o[key] = {};
        }
        return o[key];
    }, obj);

    let finalValue = value;
    if (typeof value === 'string') {
        if (value.toLowerCase() === 'true') finalValue = true;
        else if (value.toLowerCase() === 'false') finalValue = false;
        else if (!isNaN(value) && value.trim() !== '' && !isNaN(parseFloat(value))) finalValue = parseFloat(value);
    }

    target[lastKey] = finalValue;
};

const deepMerge = (target, source) => {
    for (const key in source) {
        if (source[key] instanceof Object && target[key] instanceof Object) {
            deepMerge(target[key], source[key]);
        } else {
            target[key] = source[key];
        }
    }
    return target;
};


// New function to convert array of objects to CSV string
const unparseCsv = (data) => {
    if (!data || data.length === 0) {
        return '';
    }

    // Helper to flatten nested objects and get all unique keys from the entire dataset
    const getAllKeys = (obj, prefix = '') => {
        return Object.keys(obj).reduce((acc, k) => {
            const pre = prefix.length ? prefix + '.' : '';
            if (obj[k] && typeof obj[k] === 'object' && !Array.isArray(obj[k])) {
                Object.assign(acc, getAllKeys(obj[k], pre + k));
            } else {
                acc[pre + k] = true;
            }
            return acc;
        }, {});
    };

    const allKeysObject = data.reduce((acc, row) => Object.assign(acc, getAllKeys(row)), {});
    const headers = Object.keys(allKeysObject);
    const csvRows = [headers.join(',')];

    data.forEach(row => {
        const values = headers.map(header => {
            let value = getNestedValue(row, header);
            if (value === null || value === undefined) {
                value = '';
            }
            const stringValue = String(value);
            if (stringValue.includes(',') || stringValue.includes('"') || stringValue.includes('\n')) {
                return `"${stringValue.replace(/"/g, '""')}"`;
            }
            return stringValue;
        });
        csvRows.push(values.join(','));
    });

    return csvRows.join('\n');
};

// New function to convert CSV string to array of objects
const parseCsv = (csvString) => {
    const lines = csvString.trim().split(/\r?\n/);
    if (lines.length < 2) return [];

    const headers = lines[0].split(',').map(h => h.trim());
    const data = [];

    for (let i = 1; i < lines.length; i++) {
        if (!lines[i]) continue;

        const values = lines[i].split(',');
        const obj = {};
        headers.forEach((header, index) => {
            if (values[index] !== undefined) {
                let value = values[index].trim();
                if (value.startsWith('"') && value.endsWith('"')) {
                    value = value.slice(1, -1).replace(/""/g, '"');
                }
                setNestedValue(obj, header, value);
            }
        });
        data.push(obj);
    }
    return data;
};

const DropZone = ({ onDrop, children, className, acceptedTypes, zoneId, zoneType }) => {
    const [isOver, setIsOver] = useState(false);

    const handleDragOver = (e) => {
        e.preventDefault();
        try {
            const data = JSON.parse(e.dataTransfer.getData('application/json'));
            if (acceptedTypes.includes(data.type)) {
                setIsOver(true);
                e.dataTransfer.dropEffect = 'move';
            } else {
                e.dataTransfer.dropEffect = 'none';
            }
        } catch (error) {
            e.dataTransfer.dropEffect = 'none';
        }
    };

    const handleDrop = (e) => {
        e.preventDefault();
        setIsOver(false);
        try {
            const data = JSON.parse(e.dataTransfer.getData('application/json'));
            if (acceptedTypes.includes(data.type)) {
                onDrop({
                    draggedItemId: data.id,
                    draggedItemType: data.type,
                    targetZoneId: zoneId,
                    targetZoneType: zoneType,
                });
            }
        } catch (error) {
            console.error("Drop error:", error);
        }
    };

    return (
        <div
            onDragOver={handleDragOver}
            onDragLeave={() => setIsOver(false)}
            onDrop={handleDrop}
            className={`transition-colors duration-200 rounded-lg ${className} ${isOver ? 'bg-blue-100 dark:bg-blue-900/50 ring-2 ring-blue-500' : ''}`}
        >
            {children}
        </div>
    );
};

const EditableCell = ({ item, itemType, column, onSave, className = '' }) => {
    const value = getNestedValue(item, column.key);
    const [isEditing, setIsEditing] = useState(false);
    const [editValue, setEditValue] = useState(value);
    const inputRef = useRef(null);

    useEffect(() => {
        if (isEditing && inputRef.current) {
            inputRef.current.focus();
            if (column.type === 'text' || column.type === 'number') {
                inputRef.current.select();
            }
        }
    }, [isEditing, column.type]);

    useEffect(() => {
        setEditValue(value);
    }, [value]);

    const handleSave = () => {
        if (column.type === 'number' && isNaN(editValue)) {
            return;
        }
        onSave(item, column.key, editValue, itemType);
        setIsEditing(false);
    };

    const handleCancel = () => {
        setEditValue(value);
        setIsEditing(false);
    };

    const handleKeyDown = (e) => {
        if (e.key === 'Enter') handleSave();
        if (e.key === 'Escape') handleCancel();
    };

    if (isEditing) {
        const inputClasses = "w-full px-2 py-1 border-2 border-blue-500 rounded-md focus:outline-none focus:ring-2 focus:ring-blue-500 focus:ring-offset-1 shadow-md transition-all duration-200 bg-white dark:bg-slate-700 dark:text-white dark:border-sky-400";
        if (column.type === 'select') {
            return (
                <td className={`px-4 py-2.5 text-sm ${className}`}>
                    <select
                        ref={inputRef}
                        value={editValue}
                        onChange={(e) => setEditValue(e.target.value)}
                        onBlur={handleSave}
                        onKeyDown={handleKeyDown}
                        className={inputClasses}
                    >
                        {column.options.map(opt => (
                            <option key={opt} value={opt}>{statusMap[opt]?.text || opt}</option>
                        ))}
                    </select>
                </td>
            );
        }

        if (column.type === 'boolean') {
            return (
                <td className={`px-4 py-2.5 text-sm text-center ${className}`}>
                    <input
                        ref={inputRef}
                        type="checkbox"
                        checked={!!editValue}
                        onChange={(e) => {
                            onSave(item, column.key, e.target.checked, itemType);
                            setIsEditing(false);
                        }}
                        className="w-4 h-4 text-blue-600 border-gray-300 rounded focus:ring-2 focus:ring-blue-500 dark:bg-slate-600 dark:border-slate-500"
                    />
                </td>
            );
        }

        return (
            <td className={`px-4 py-2.5 text-sm ${className}`}>
                <input
                    ref={inputRef}
                    type={column.type === 'date' ? 'date' : 'text'}
                    value={editValue || ''}
                    onChange={(e) => setEditValue(e.target.value)}
                    onBlur={handleSave}
                    onKeyDown={handleKeyDown}
                    className={inputClasses}
                />
            </td>
        );
    }

    return (
        <td className={`px-4 py-2.5 text-sm text-slate-800 dark:text-slate-200 font-medium whitespace-nowrap group ${className}`}>
            <div className="flex items-center gap-2">
                <span>{column.type === 'boolean' ? (value ? 'כן' : 'לא') : (value || '-')}</span>
                <button
                    onClick={() => setIsEditing(true)}
                    className="opacity-0 group-hover:opacity-100 transition-opacity duration-200"
                >
                    <Icon path={ICONS.edit} className="w-4 h-4 text-gray-400 hover:text-blue-600 dark:hover:text-sky-400 transition-colors duration-200" />
                </button>
            </div>
        </td>
    );
};

        if (!newColumn.groupName || !newColumn.label) {
            console.error("יש לבחור קבוצה ולציין שם עמודה.");
            return;
        }

        const newKey = `custom_${newColumn.label.trim().toLowerCase().replace(/\s+/g, '_')}`;
        const newColumnConfig = {
            key: newKey,
            label: newColumn.label.trim(),
            editable: true,
            type: newColumn.type,
            visible: true
        };

        const updatedConfig = localConfig.map(group => {
            if (group.name === newColumn.groupName) {
                return { ...group, columns: [...group.columns, newColumnConfig] };
            }
            return group;
        });

// --- NEW: Strategic Dashboard Modal (Replaces AssignmentManagerModal) ---
const StrategicDashboardModal = ({ isOpen, onClose, data, onApplyChanges, allOwners, addNotification }) => {
    if (!isOpen) return null;

    const [localData, setLocalData] = useState(JSON.parse(JSON.stringify(data)));
    const [scenarioSignedOwners, setScenarioSignedOwners] = useState(new Set());
    const [scenarioName, setScenarioName] = useState('תרחיש נוכחי');
    const [savedScenarios, setSavedScenarios] = useState([]); // For saving multiple scenarios

    const majorityStatus = SpecialMajorityCalculator.analyze(localData.buildings, scenarioSignedOwners);

    const toggleOwnerSignature = (ownerIdNumber) => {
        setScenarioSignedOwners(prev => {
            const newSet = new Set(prev);
            if (newSet.has(ownerIdNumber)) {
                newSet.delete(ownerIdNumber);
            } else {
                newSet.add(ownerIdNumber);
            }
            return newSet;
        });
    };

    const saveScenario = () => {
        const newScenario = {
            name: scenarioName,
            signedOwners: Array.from(scenarioSignedOwners),
            status: majorityStatus
        };
        setSavedScenarios(prev => [...prev, newScenario]);
        addNotification(`תרחיש "${scenarioName}" נשמר בהצלחה`);
    };

    const loadScenario = (scenario) => {
        setScenarioSignedOwners(new Set(scenario.signedOwners));
        setScenarioName(scenario.name);
    };

    const handleAssign = (dropEvent) => {
        setLocalData(prev => {
            const newLocal = JSON.parse(JSON.stringify(prev));
            if (dropEvent.draggedItemType === 'unit' && dropEvent.targetZoneType === 'building') {
                const unitIndex = newLocal.unassignedUnits.findIndex(u => u.id === dropEvent.draggedItemId);
                if (unitIndex > -1) {
                    const [unit] = newLocal.unassignedUnits.splice(unitIndex, 1);
                    const building = newLocal.buildings.find(b => b.id === dropEvent.targetZoneId);
                    if (building) building.units.push(unit);
                }
            } else if (dropEvent.draggedItemType === 'owner' && dropEvent.targetZoneType === 'unit') {
                let owner;
                let ownerIndex;
                let ownerUnit;
                newLocal.buildings.forEach(b => b.units.forEach(u => {
                    ownerIndex = u.owners.findIndex(o => o.id === dropEvent.draggedItemId);
                    if (ownerIndex > -1) {
                        owner = u.owners[ownerIndex];
                        ownerUnit = u;
                    }
                }));
                if (!owner && newLocal.unassignedOwners) {
                    ownerIndex = newLocal.unassignedOwners.findIndex(o => o.id === dropEvent.draggedItemId);
                    if (ownerIndex > -1) owner = newLocal.unassignedOwners[ownerIndex];
                }
                if (owner) {
                    if (ownerUnit) {
                        ownerUnit.owners.splice(ownerIndex, 1);
                    } else if (newLocal.unassignedOwners) {
                        newLocal.unassignedOwners.splice(ownerIndex, 1);
                    }
                    const targetUnit = newLocal.buildings.flatMap(b => b.units).find(u => u.id === dropEvent.targetZoneId) || newLocal.unassignedUnits.find(u => u.id === dropEvent.targetZoneId);
                    if (targetUnit) targetUnit.owners.push(owner);
                }
            }
            return newLocal;
        });
    };

    const applyChanges = () => {
        onApplyChanges(localData);
        onClose();
    };

    return (
        <div className="fixed inset-0 bg-black/60 backdrop-blur-sm z-50 flex justify-center items-center p-4 animate-fadeIn" onClick={onClose}>
            <div className="bg-white dark:bg-slate-800 rounded-2xl shadow-2xl w-full max-w-6xl max-h-[90vh] overflow-hidden animate-slideUp flex flex-col" onClick={e => e.stopPropagation()}>
                <div className="sticky top-0 bg-white/80 dark:bg-slate-800/80 backdrop-blur-md p-6 border-b border-slate-200 dark:border-slate-700 flex justify-between items-center">
                    <h2 className="text-2xl font-bold text-slate-900 dark:text-slate-100">סדנת תכנון אסטרטגי</h2>
                    <button onClick={onClose} className="p-2 rounded-full hover:bg-slate-100 dark:hover:bg-slate-700 transition-colors duration-200">
                        <Icon path={ICONS.xMark} className="w-6 h-6 text-slate-600 dark:text-slate-300" />
                    </button>
                </div>
                <div className="p-6 overflow-y-auto flex-grow space-y-6">
                    <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
                        <div className="space-y-4">
                            <h3 className="text-lg font-semibold">שיוך יחידות ובניינים</h3>
                            <div className="grid grid-cols-2 gap-4">
                                <DropZone onDrop={handleAssign} acceptedTypes={['unit', 'owner']} zoneId="unassigned" zoneType="unassigned">
                                    <h4>לא משויך</h4>
                                    {localData.unassignedUnits.map(u => <DraggableItem key={u.id} item={u} type="unit" />)}
                                    {localData.unassignedOwners.map(o => <DraggableItem key={o.id} item={o} type="owner" />)}
                                </DropZone>
                                <div>
                                    <h4>בניינים</h4>
                                    {localData.buildings.map(b => (
                                        <DropZone key={b.id} onDrop={handleAssign} acceptedTypes={['unit']} zoneId={b.id} zoneType="building">
                                            <h5>{b.name}</h5>
                                            {b.units.map(u => <DraggableItem key={u.id} item={u} type="unit" />)}
                                        </DropZone>
                                    ))}
                                </div>
                            </div>
                        </div>
                        <div className="space-y-4">
                            <h3 className="text-lg font-semibold">סימולציית חתימות</h3>
                            <input value={scenarioName} onChange={e => setScenarioName(e.target.value)} placeholder="שם תרחיש" className="w-full p-2 border rounded" />
                            <button onClick={saveScenario} className="px-4 py-2 bg-blue-500 text-white rounded">שמור תרחיש</button>
                            <div className="h-64 overflow-y-auto space-y-2">
                                {allOwners.map(owner => (
                                    <div key={owner.id} className="flex items-center gap-2 p-2 bg-slate-100 rounded">
                                        <input type="checkbox" checked={scenarioSignedOwners.has(owner.idNumber)} onChange={() => toggleOwnerSignature(owner.idNumber)} />
                                        <span>{owner.name}</span>
                                    </div>
                                ))}
                            </div>
                        </div>
                    </div>
                    <div>
                        <h3 className="text-lg font-semibold">תוצאות סימולציה</h3>
                        <p>רוב מיוחס כללי: {majorityStatus.cluster.agreement.overallMet ? 'עומד' : 'לא עומד'}</p>
                        <p>דירות חתומות: {majorityStatus.cluster.agreement.apartment.actual} / {majorityStatus.cluster.agreement.apartment.required}</p>
                    </div>
                    <div className="space-y-2">
                        <h3 className="text-lg font-semibold">תרחישים שמורים</h3>
                        {savedScenarios.map((sc, idx) => (
                            <div key={idx} className="flex justify-between p-2 bg-slate-100 rounded">
                                <span>{sc.name}</span>
                                <button onClick={() => loadScenario(sc)} className="text-blue-500">טען</button>
                            </div>
                        ))}
                    </div>
                </div>
                <div className="sticky bottom-0 bg-slate-50 dark:bg-slate-800/50 p-6 border-t border-slate-200 dark:border-slate-700 flex justify-end gap-3">
                    <button onClick={onClose} className="px-6 py-2.5 bg-slate-200 text-slate-700 rounded-lg">ביטול</button>
                    <button onClick={applyChanges} className="px-6 py-2.5 bg-blue-500 text-white rounded-lg">החל שינויים</button>
                </div>
            </div>
        </div>
    );
};

// --- MAIN APP COMPONENT ---
export default function App() {
    const [focusLevel, setFocusLevel] = useState('owner');
    const [searchTerm, setSearchTerm] = useState('');
    const [sortConfig, setSortConfig] = useState(null);
    const [selectedItem, setSelectedItem] = useState(null);
    const [selectedItemType, setSelectedItemType] = useState(null);
    const [showActivityLog, setShowActivityLog] = useState(false);
    const [showColumnManager, setShowColumnManager] = useState(false);
    const [showStrategyDashboard, setShowStrategyDashboard] = useState(false);
    const [data, setData] = useState(PROJECT_DATA);
    const [headerConfig, setHeaderConfig] = useState(INITIAL_HEADER_CONFIG);
    const [theme, setTheme] = useState('light');
    const [notifications, setNotifications] = useState([]);
    // NEW: State for confirmation modal
    const [confirmation, setConfirmation] = useState({ isOpen: false });


    useEffect(() => {
        const savedTheme = localStorage.getItem('theme');
        const systemPrefersDark = window.matchMedia('(prefers-color-scheme: dark)').matches;
        setTheme(savedTheme === 'dark' || (!savedTheme && systemPrefersDark) ? 'dark' : 'light');
    }, []);

    useEffect(() => {
        document.documentElement.classList.toggle('dark', theme === 'dark');
        localStorage.setItem('theme', theme);
    }, [theme]);

    // Effect to reset search and sort when focus level changes
    useEffect(() => {
        setSearchTerm('');
        setSortConfig(null);
    }, [focusLevel]);


    const toggleTheme = () => setTheme(prev => prev === 'light' ? 'dark' : 'light');

    const addNotification = useCallback((message) => {
        const newNotif = { id: Date.now(), message };
        setNotifications(prev => [...prev, newNotif]);
        setTimeout(() => setNotifications(prev => prev.filter(n => n.id !== newNotif.id)), 5000);
    }, []);

    const ALL_OWNERS_WITH_CONTEXT = useMemo(() => {
        const assignedOwners = data.buildings.flatMap(b =>
            b.units.flatMap(u =>
                u.owners.map(o => ({
                    ...o,
                    signatureStatus: getCalculatedSignatureStatus(o),
                    unit: u,
                    building: b
                }))
            )
        );
        const unassigned = (data.unassignedOwners || []).map(o => ({ ...o, signatureStatus: getCalculatedSignatureStatus(o), unit: null, building: null }));
        return [...assignedOwners, ...unassigned];
    }, [data]);

    const ALL_UNITS_WITH_CONTEXT = useMemo(() => {
        const assignedUnits = data.buildings.flatMap(b => b.units.map(u => ({ ...u, building: b })));
        const unassigned = (data.unassignedUnits || []).map(u => ({ ...u, building: null }));

        return [...assignedUnits, ...unassigned].map(u => {
            const signedOwners = u.owners.filter(o => getCalculatedSignatureStatus(o) === 'signed').length;
            const docStatuses = Object.values(u.requiredDocuments || {}).map(d => d.status);
            const okDocs = docStatuses.filter(s => s === 'ok' || s === 'signed').length;
            const relevantDocs = docStatuses.filter(s => s !== 'na').length;
            return {
                ...u,
                name: u.identification.fullAddress,
                ownerNames: u.owners.map(o => o.name).join(', '),
                ownerCount: u.owners.length,
                signingStatus: u.owners.length > 0 ? (signedOwners / u.owners.length) * 100 : 0,
                documentsStatus: relevantDocs > 0 ? (okDocs / relevantDocs) * 100 : 0,
            }
        });
    }, [data]);

    const ALL_BUILDINGS_WITH_CONTEXT = useMemo(() => {
        const initialSignedOwners = new Set(data.buildings.flatMap(b => b.units.flatMap(u => u.owners)).filter(o => getCalculatedSignatureStatus(o) === 'signed').map(o => o.idNumber));
        const majorityStatus = SpecialMajorityCalculator.analyze(data.buildings, initialSignedOwners);

        return data.buildings.map(b => {
            const buildingDetail = majorityStatus.buildings[b.id];
            const signedUnits = b.units.filter(u => u.owners.every(o => getCalculatedSignatureStatus(o) === 'signed')).length;

            return {
                ...b,
                unitCount: b.units.length,
                specialMajorityStatus: buildingDetail?.agreement.overallMet ? 'approved' : (buildingDetail?.isValidForCalc ? 'failing' : 'invalid'),
                signingProgress: b.units.length > 0 ? (signedUnits / b.units.length) * 100 : 0,
                commonPropertyProgress: buildingDetail?.totalCommonProperty > 0 ? (buildingDetail.signedCommonProperty / buildingDetail.totalCommonProperty) * 100 : 0,
                financialIssuesCount: b.units.filter(u => u.legal.liens !== 'נקי').length,
            };
        });
    }, [data]);

    const PROJECT_WITH_CONTEXT = useMemo(() => {
        const initialSignedOwners = new Set(data.buildings.flatMap(b => b.units.flatMap(u => u.owners)).filter(o => getCalculatedSignatureStatus(o) === 'signed').map(o => o.idNumber));
        const majorityStatus = SpecialMajorityCalculator.analyze(data.buildings, initialSignedOwners);
        const allUnits = data.buildings.flatMap(b => b.units);
        const allOwners = new Set(allUnits.flatMap(u => u.owners.map(o => o.idNumber))).size;

        return [{
            ...data,
            totalUnits: allUnits.length,
            totalOwners: allOwners,
            overallSpecialMajorityStatus: majorityStatus.cluster.agreement.overallMet ? 'approved' : 'failing',
            overallSigningProgress: majorityStatus.cluster.totalApartments > 0 ? (majorityStatus.cluster.agreement.apartment.actual / majorityStatus.cluster.totalApartments) * 100 : 0,
        }];
    }, [data]);

    const updateData = useCallback((item, field, value, level) => {
        setData(prevData => {
            const newData = JSON.parse(JSON.stringify(prevData));
            let itemToUpdate;
            let identifier;

            if (level === 'owner') {
                let found = false;
                for (const b of newData.buildings) {
                    for (const u of b.units) {
                        const owner = u.owners.find(o => o.id === item.id);
                        if (owner) {
                            itemToUpdate = owner;
                            found = true;
                            break;
                        }
                    }
                    if (found) break;
                }
                if (!found && newData.unassignedOwners) {
                    itemToUpdate = newData.unassignedOwners.find(o => o.id === item.id);
                }

                identifier = item.name;
            } else if (level === 'unit') {
                let foundInBuilding = false;
                for (const building of newData.buildings) {
                    const unit = building.units.find(u => u.id === item.id);
                    if (unit) {
                        itemToUpdate = unit;
                        foundInBuilding = true;
                        break;
                    }
                }
                if (!foundInBuilding && newData.unassignedUnits) {
                    itemToUpdate = newData.unassignedUnits.find(u => u.id === item.id);
                }
                identifier = `Unit ${item.subplot}`;
            } else if (level === 'building') {
                itemToUpdate = newData.buildings.find(b => b.id === item.id);
                identifier = item.name;
            } else if (level === 'project') {
                itemToUpdate = newData;
                identifier = item.projectName;
            }

            if (itemToUpdate) {
                const oldValue = getNestedValue(itemToUpdate, field);
                setNestedValue(itemToUpdate, field, value);
                logActivity('updated', level, identifier, { fieldChanged: field, oldValue, newValue: value });
            }
            return newData;
        });
    }, []);

    const handleConfigAndUpdateData = useCallback((level, newConfig) => {
        const oldConfig = headerConfig[level];
        setHeaderConfig(prev => ({ ...prev, [level]: newConfig }));

        const getColumns = (config) => new Set(config.flatMap(g => g.columns.map(c => c.key)));

        const oldCols = getColumns(oldConfig);
        const newCols = getColumns(newConfig);
        const addedCols = [...newCols].filter(key => !oldCols.has(key));

        if (addedCols.length > 0) {
            setData(prevData => {
                const newData = JSON.parse(JSON.stringify(prevData));
                const itemsToUpdate = [];

                if (level === 'owner') {
                    newData.buildings.forEach(b => b.units.forEach(u => itemsToUpdate.push(...u.owners)));
                    itemsToUpdate.push(...(newData.unassignedOwners || []));
                } else if (level === 'unit') {
                    newData.buildings.forEach(b => itemsToUpdate.push(...b.units));
                    itemsToUpdate.push(...(newData.unassignedUnits || []));
                } else if (level === 'building') {
                    itemsToUpdate.push(...newData.buildings);
                } else if (level === 'project') {
                    itemsToUpdate.push(newData);
                }

                itemsToUpdate.forEach(item => {
                    addedCols.forEach(key => {
                        if (!getNestedValue(item, key)) {
                            const colConfig = newConfig.flatMap(g => g.columns).find(c => c.key === key);
                            let defaultValue = null;
                            if (colConfig?.type === 'status') defaultValue = (colConfig.options || ['missing'])[0];
                            if (colConfig?.type === 'boolean') defaultValue = false;
                            if (colConfig?.type === 'number') defaultValue = 0;
                            if (colConfig?.type === 'date') defaultValue = new Date().toISOString().split('T')[0];
                            if (colConfig?.type === 'text') defaultValue = '';
                            if (colConfig?.type === 'select') defaultValue = (colConfig.options || [])[0] || '';

                            const path = key.split('.').slice(0, -1).join('.');
                            const finalKey = key.split('.').pop();
                            const target = path ? (getNestedValue(item, path) || setNestedValue(item, path, {})) : item;
                            target[finalKey] = defaultValue;
                        }
                    });
                });
                return newData;
            });
        }
    }, [headerConfig]);

    const currentTableData = useMemo(() => {
        if (focusLevel === 'owner') return ALL_OWNERS_WITH_CONTEXT;
        if (focusLevel === 'unit') return ALL_UNITS_WITH_CONTEXT;
        if (focusLevel === 'building') return ALL_BUILDINGS_WITH_CONTEXT;
        if (focusLevel === 'project') return PROJECT_WITH_CONTEXT;
        return [];
    }, [focusLevel, data, ALL_OWNERS_WITH_CONTEXT, ALL_UNITS_WITH_CONTEXT, ALL_BUILDINGS_WITH_CONTEXT, PROJECT_WITH_CONTEXT]);

    const sortedAndFilteredData = useMemo(() => {
        let processedData = [...currentTableData];

        if (searchTerm) {
            processedData = processedData.filter(item => {
                const normalizedSearch = searchTerm.toLowerCase();
                return Object.values(item).some(val => 
                    String(val).toLowerCase().includes(normalizedSearch)
                ) || (item.identification && Object.values(item.identification).some(val => String(val).toLowerCase().includes(normalizedSearch)));
            });
        }

        if (sortConfig !== null) {
            processedData.sort((a, b) => {
                const valA = getNestedValue(a, sortConfig.key);
                const valB = getNestedValue(b, sortConfig.key);
                const dir = sortConfig.direction === 'ascending' ? 1 : -1;

                if (valA == null) return 1 * dir;
                if (valB == null) return -1 * dir;

                if (typeof valA === 'string') {
                    return valA.localeCompare(valB, 'he') * dir;
                }
                if (typeof valA === 'number') {
                    return (valA - valB) * dir;
                }
                return 0;
            });
        }
        return processedData;
    }, [currentTableData, searchTerm, sortConfig, focusLevel]);

    const requestSort = (key, type) => {
        let direction = 'ascending';
        if (sortConfig?.key === key && sortConfig.direction === 'ascending') {
            direction = 'descending';
        }
        setSortConfig({ key, direction, type });
    };

    const getSortIcon = (key) => {
        if (sortConfig?.key !== key) return ICONS.sort;
        return sortConfig.direction === 'ascending' ? ICONS.sortAsc : ICONS.sortDesc;
    }

    const kpis = useMemo(() => {
        const allUnitsInBuildings = data.buildings.flatMap(b => b.units);
        const allUnitsWithUnassigned = [...allUnitsInBuildings, ...(data.unassignedUnits || [])];

        const allOwnersInBuildings = allUnitsInBuildings.flatMap(u => u.owners);
        const allOwnersWithUnassigned = [...allOwnersInBuildings, ...(data.unassignedOwners || [])];

        const totalOwners = new Set(allOwnersWithUnassigned.map(o => o.idNumber)).size;

        const allOwnersForIcons = allOwnersWithUnassigned.map(o => ({ 
            id: o.id, 
            tooltip: `${o.name} (ת.ז. ${o.idNumber})`, 
            type: 'owner',
            status: getCalculatedSignatureStatus(o)
        }));
        const allUnitsForIcons = allUnitsWithUnassigned.map(u => ({ id: u.id, tooltip: u.identification.fullAddress, type: 'unit' }));
        const allBuildingsForIcons = data.buildings.map(b => ({ id: b.id, tooltip: b.name, type: 'building' }));

        return {
            totalOwners,
            totalUnits: allUnitsWithUnassigned.length,
            totalBuildings: data.buildings.length,
            allOwnersForIcons,
            allUnitsForIcons,
            allBuildingsForIcons,
        };
    }, [data]);

    const handleAddRow = () => {
        setData(prevData => {
            const newData = JSON.parse(JSON.stringify(prevData));
            if (focusLevel === 'owner') {
                const newOwner = createOwnerData(null, { name: "בעלים חדש" });
                if (!newData.unassignedOwners) newData.unassignedOwners = [];
                newData.unassignedOwners.push(newOwner);
                logActivity('created', 'owner', newOwner.id);
                addNotification(`נוסף בעלים חדש: ${newOwner.name}`);
            } else if (focusLevel === 'unit') {
                const allSubplots = [...newData.buildings.flatMap(b => b.units.map(u => u.subplot)), ...(newData.unassignedUnits || []).map(u => u.subplot)];
                const maxSubplot = Math.max(0, ...allSubplots.filter(s => s));
                const newUnit = {
                    id: `U${Date.now()}`,
                    subplot: maxSubplot + 1,
                    propertyType: 'דירת מגורים',
                    identification: { fullAddress: `יחידה חדשה ${maxSubplot + 1}` },
                    physical: {},
                    legal: { commonPropertyPercentage: 0 },
                    status: {},
                    requiredDocuments: {},
                    owners: []
                };
                if (!newData.unassignedUnits) newData.unassignedUnits = [];
                newData.unassignedUnits.push(newUnit);
                logActivity('created', 'unit', `Unit ${maxSubplot + 1}`);
                addNotification(`נוספה יחידה חדשה: ${newUnit.identification.fullAddress}`);
            } else if (focusLevel === 'building') {
                const newId = `B${Date.now()}`;
                const newBuilding = {
                    id: newId,
                    name: `בניין חדש ${newData.buildings.length + 1}`,
                    gush: '00000',
                    helka: '000',
                    units: [],
                    buildingType: 'מגורים',
                    constructionStatus: 'not-started',
                    physicalCharacteristics: { age: 0, numFloors: 0, numEntrances: 1 },
                    condominiumInfo: { activeResidentsCommittee: { exists: false } }
                };
                newData.buildings.push(newBuilding);
                logActivity('created', 'building', newId);
                addNotification(`נוסף בניין חדש: ${newBuilding.name}`);
            }
            return newData;
        });
    };

    // NEW: Delete handler to open confirmation
    const handleDeleteRow = (item, level) => {
        setConfirmation({
            isOpen: true,
            title: `מחיקת ${focusLevel === 'owner' ? 'בעלים' : focusLevel === 'unit' ? 'יחידה' : 'בניין'}`,
            message: `האם אתה בטוח שברצונך למחוק את '${item.name || item.id}'? פעולה זו אינה הפיכה.`,
            onConfirm: () => confirmDelete(item, level),
        });
    };

    // NEW: Actual deletion logic
    const confirmDelete = (item, level) => {
        setData(prevData => {
            let newData = JSON.parse(JSON.stringify(prevData));
            if (level === 'owner') {
                let found = false;
                newData.buildings.forEach(b => {
                    b.units.forEach(u => {
                        const index = u.owners.findIndex(o => o.id === item.id);
                        if (index > -1) {
                            u.owners.splice(index, 1);
                            found = true;
                        }
                    });
                });
                if (!found && newData.unassignedOwners) {
                     const index = newData.unassignedOwners.findIndex(o => o.id === item.id);
                     if (index > -1) newData.unassignedOwners.splice(index, 1);
                }
            } else if (level === 'unit') {
                 newData.buildings.forEach(b => {
                     const index = b.units.findIndex(u => u.id === item.id);
                     if (index > -1) b.units.splice(index, 1);
                 });
                 if (newData.unassignedUnits) {
                     const index = newData.unassignedUnits.findIndex(u => u.id === item.id);
                     if (index > -1) newData.unassignedUnits.splice(index, 1);
                 }
            } else if (level === 'building') {
                 const index = newData.buildings.findIndex(b => b.id === item.id);
                 if (index > -1) {
                     const [removedBuilding] = newData.buildings.splice(index, 1);
                     // Move units from deleted building to unassigned
                     if (!newData.unassignedUnits) newData.unassignedUnits = [];
                     newData.unassignedUnits.push(...removedBuilding.units);
                 }
            }
            logActivity('deleted', level, item.name || item.id);
            addNotification(`'${item.name || item.id}' נמחק בהצלחה.`);
            return newData;
        });
        setConfirmation({ isOpen: false }); // Close modal after action
    };

    const assignUnitToBuilding = (unitId, buildingId) => {
        setData(prevData => {
            const newData = JSON.parse(JSON.stringify(prevData));
            const unitIndex = (newData.unassignedUnits || []).findIndex(u => u.id === unitId);
            if (unitIndex === -1) return prevData;

            const [unitToAssign] = newData.unassignedUnits.splice(unitIndex, 1);
            const targetBuilding = newData.buildings.find(b => b.id === buildingId);
            if (targetBuilding) {
                targetBuilding.units.push(unitToAssign);
            }
            logActivity('updated', 'building', `Assigned unit ${unitId} to building ${buildingId}`);
            addNotification(`שויכה יחידה ${unitId} לבניין ${buildingId}`);
            return newData;
        });
    };

    const unassignUnitFromBuilding = (unitId, buildingId) => {
        setData(prevData => {
            const newData = JSON.parse(JSON.stringify(prevData));
            const sourceBuilding = newData.buildings.find(b => b.id === buildingId);
            if (!sourceBuilding) return prevData;

            const unitIndex = sourceBuilding.units.findIndex(u => u.id === unitId);
            if (unitIndex === -1) return prevData;

            const [unitToUnassign] = sourceBuilding.units.splice(unitIndex, 1);
            if (!newData.unassignedUnits) newData.unassignedUnits = [];
            newData.unassignedUnits.push(unitToUnassign);
            logActivity('updated', 'building', `Unassigned unit ${unitId} from building ${buildingId}`);
            addNotification(`בוטל שיוך יחידה ${unitId} מבניין ${buildingId}`);
            return newData;
        });
    };

    const handleExport = () => {
        const csv = unparseCsv(currentTableData);
        const blob = new Blob(["\uFEFF" + csv], { type: 'text/csv;charset=utf-8;' });
        const url = URL.createObjectURL(blob);
        const link = document.createElement('a');
        link.href = url;
        link.download = `${focusLevel}_data.csv`;
        link.click();
        addNotification('נתונים יוצאו בהצלחה');
    };

    const handleImport = (e) => {
        const file = e.target.files[0];
        if (!file) return;

        const reader = new FileReader();
        reader.onload = (event) => {
            const csvString = event.target.result;
            const importedData = parseCsv(csvString);

            setData(prevData => {
                const newData = JSON.parse(JSON.stringify(prevData));

                if (focusLevel === 'owner') {
                    importedData.forEach(importedOwner => {
                        let ownerFound = false;
                        for (const b of newData.buildings) {
                            for (const u of b.units) {
                                const ownerIndex = u.owners.findIndex(o => o.idNumber === importedOwner.idNumber);
                                if (ownerIndex > -1) {
                                    u.owners[ownerIndex] = deepMerge(u.owners[ownerIndex], importedOwner);
                                    ownerFound = true;
                                    break;
                                }
                            }
                            if (ownerFound) break;
                        }
                        if (!ownerFound && newData.unassignedOwners) {
                            const ownerIndex = newData.unassignedOwners.findIndex(o => o.idNumber === importedOwner.idNumber);
                            if (ownerIndex > -1) {
                                newData.unassignedOwners[ownerIndex] = deepMerge(newData.unassignedOwners[ownerIndex], importedOwner);
                                ownerFound = true;
                            }
                        }
                        if (!ownerFound) {
                            if (!newData.unassignedOwners) newData.unassignedOwners = [];
                            newData.unassignedOwners.push(createOwnerData(null, importedOwner));
                        }
                    });
                }
                // TODO: Implement import logic for other focus levels
                return newData;
            });
            addNotification('ייבוא נתונים הושלם');
        };
        reader.readAsText(file);
    };

    return (
        <div className="min-h-screen bg-slate-50 dark:bg-slate-900 font-sans text-right transition-colors duration-500" dir="rtl">
            <DetailModal
                item={selectedItem}
                itemType={selectedItemType}
                onClose={() => setSelectedItem(null)}
                onUpdate={updateData}
                unassignedUnits={data.unassignedUnits || []}
                onAssignUnit={assignUnitToBuilding}
                onUnassignUnit={unassignUnitFromBuilding}
            />
            <ActivityLogModal isOpen={showActivityLog} onClose={() => setShowActivityLog(false)} />
            <ColumnManagerModal
                isOpen={showColumnManager}
                onClose={() => setShowColumnManager(false)}
                headerConfig={headerConfig}
                focusLevel={focusLevel}
                onUpdateConfigAndData={handleConfigAndUpdateData}
            />
            {/* NEW: Replaced AssignmentManagerModal with StrategicDashboardModal */}
            <StrategicDashboardModal
                isOpen={showStrategyDashboard}
                onClose={() => setShowStrategyDashboard(false)}
                data={data}
                onApplyChanges={setData}
                allOwners={ALL_OWNERS_WITH_CONTEXT}
                addNotification={addNotification}
            />

            <NotificationsPanel notifications={notifications} />
             {/* NEW: Render confirmation modal */}
            <ConfirmationModal 
                isOpen={confirmation.isOpen}
                onClose={() => setConfirmation({ isOpen: false })}
                onConfirm={confirmation.onConfirm}
                title={confirmation.title}
                message={confirmation.message}
            />


            <div className="mx-auto px-4 sm:px-6 lg:px-8 py-10">
                <header className="relative text-center mb-8">
                    <div className="absolute top-0 right-0 flex gap-2">
                        <button
                            onClick={() => setShowActivityLog(true)}
                            title="יומן פעילות"
                            className="p-2.5 rounded-lg text-slate-500 dark:text-slate-400 hover:bg-slate-100 dark:hover:bg-slate-700"
                        >
                            <Icon path={ICONS.history} />
                        </button>
                        <ThemeToggle theme={theme} toggleTheme={toggleTheme} />
                    </div>
                    <h1 className="text-4xl font-black text-transparent bg-clip-text bg-gradient-to-r from-blue-600 to-indigo-600 dark:from-blue-400 dark:to-indigo-400 tracking-tight">
                        מערכת ניהול פרויקטים
                    </h1>
                    <p className="mt-3 text-lg text-slate-600 dark:text-slate-300">ניהול מתקדם לפרויקטי התחדשות עירונית</p>
                </header>

                <FocusNavigation focusLevel={focusLevel} setFocusLevel={setFocusLevel} />

                <main className="space-y-8">
                    <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-6">
                        <SummaryCard
                            label="בעלי זכויות"
                            value={kpis.totalOwners}
                            items={kpis.allOwnersForIcons}
                            iconPath={ICONS.users}
                            iconColorClass={{ bg: 'bg-blue-100 dark:bg-blue-900/50', text: 'text-blue-600 dark:text-blue-400' }}
                            type="owner"
                        />
                        <SummaryCard
                            label="יחידות"
                            value={kpis.totalUnits}
                             items={kpis.allUnitsForIcons}
                            iconPath={ICONS.unit}
                            iconColorClass={{ bg: 'bg-yellow-100 dark:bg-yellow-900/50', text: 'text-yellow-600 dark:text-yellow-400' }}
                             type="unit"
                        />
                        <SummaryCard
                            label="בניינים"
                            value={kpis.totalBuildings}
                             items={kpis.allBuildingsForIcons}
                            iconPath={ICONS.building}
